\`\`\`bash

git clone https://github.com/SigmaSauer07/alsania-wallet
cd alsania-wallet
git checkout cursor/develop-and-test-cross-platform-smart-wallet-with-advanced-nft-features-b026

# 2. Verify
bash verify-deployment.sh

# 3. Build
make build

# 4. Test
make test

# 5. Deploy
make deploy-production
\`\`\`
